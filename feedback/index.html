import { apiRequest, hasAdminSession, getAdminToken, sanitizeText, formatNumber } from '../shared.js'; // ── SYSTEM METADATA HELPER ──────────────────────────────────────────────── /** Detects basic browser and device info to attach to feedback reports */ function getSystemMeta() { const ua = navigator.userAgent; let browser = "Unknown"; let device = "Desktop"; if (ua.includes("Firefox")) browser = "Firefox"; else if (ua.includes("Chrome")) browser = "Chrome"; else if (ua.includes("Safari")) browser = "Safari"; else if (ua.includes("Edge")) browser = "Edge"; if (/Mobi|Android|iPhone|iPad/i.test(ua)) device = "Mobile/Tablet"; return { browser, device, appVersion: "1.0.0", sourceApplication: window.location.pathname }; } // ── USER SUBMISSION HANDLER ────────────────────────────────────────────── /** * Submits user feedback to the backend worker. * @param {Object} formData - { type, title, description } */ export async function submitUserFeedback({ type, title, description }) { const meta = getSystemMeta(); const payload = { type: sanitizeText(type), title: sanitizeText(title), description: sanitizeText(description), ...meta }; return await apiRequest("/feedback/submit", { method: "POST", body: JSON.stringify(payload) }); } // ── ADMIN FEEDBACK MANAGEMENT ───────────────────────────────────────────── /** * Fetches all feedback submissions (Admin only). */ export async function fetchAllFeedback() { if (!hasAdminSession()) { throw new Error("Admin authorization required to fetch feedback."); } const res = await apiRequest("/feedback/list", { method: "GET" }); return res.items || []; } /** * Deletes a single feedback record by ID (Admin only). * @param {string} id - The ID of the feedback entry to delete */ export async function deleteFeedback(id) { if (!hasAdminSession()) { throw new Error("Admin authorization required to delete feedback."); } return await apiRequest("/feedback/delete", { method: "POST", body: JSON.stringify({ id, token: getAdminToken() }) }); } // ── UI CONTROLLER & RENDERING HELPERS ───────────────────────────────────── /** * Renders the admin feedback table/list into a target container element. * @param {HTMLElement|string} container - DOM element or query selector */ export async function renderAdminFeedbackPanel(container) { const target = typeof container === "string" ? document.querySelector(container) : container; if (!target) return; target.innerHTML = `
Loading feedback...
`; try { const items = await fetchAllFeedback(); if (items.length === 0) { target.innerHTML = `
No feedback entries found.
`; return; } let html = `

User Feedback (${formatNumber(items.length)})

`; items.forEach(item => { html += `
${sanitizeText(item.type)} ${sanitizeText(item.title)} ${new Date(item.timestamp).toLocaleString()}

${sanitizeText(item.description)}

Device: ${sanitizeText(item.device)} | Browser: ${sanitizeText(item.browser)} | Source: ${sanitizeText(item.sourceApplication)}
`; }); html += `
`; target.innerHTML = html; // Attach event delegation for delete buttons target.addEventListener("click", async (e) => { if (e.target.dataset.action === "delete") { const id = e.target.dataset.id; if (confirm("Are you sure you want to delete this feedback item?")) { try { await deleteFeedback(id); const card = target.querySelector(`.feedback-card[data-id="${id}"]`); if (card) card.remove(); } catch (err) { alert(`Failed to delete: ${err.message}`); } } } }); } catch (err) { target.innerHTML = `
Error loading feedback: ${sanitizeText(err.message)}
`; } }
